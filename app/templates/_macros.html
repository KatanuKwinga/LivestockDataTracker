{# A "macro" is a reusable piece of template, like a function. This one draws one form field the Bootstrap way: label, input box, and any error messages underneath in red. Written once here, it's used by every form in the app. #} {% macro render_field(field) %}
{{ field.label(class="form-label") }} {# "is-invalid" turns the box red when the field has errors. #} {{ field(class="form-control" + (" is-invalid" if field.errors else ""), **kwargs) }} {% for error in field.errors %}
{{ error }}
{% endfor %}
{% endmacro %}
